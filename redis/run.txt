#!/bin/bash
docker rm -f redis
docker network rm redisnet
docker network create redisnet
docker run  --name redis 
       --network redisnet \
       -p 6379:6379 \
       -e REDIS_PORT=6379 \
       -e REDIS_HOST=localhost\
       -e REDIS_SSL=false \
       -e ALLOW_EMPTY_PASSWORD=yes \
       -e REDIS_AOF_ENABLED=no \
       --pull always \
       -d \
      redis:8.0
