/*
 * Backward compatibility alias: listLibrary → listLibraries
 * This thin wrapper re-exports the canonical plural tool under the legacy singular name.
 * Deprecation plan: remove in a future major version once all clients migrate.
 */
import listLibraries from './listLibrary.js';

function listLibraryAlias() {
  const spec = listLibraries();
  return { ...spec, name: 'listLibrary', description: spec.description + '\n\n[Alias] Deprecated singular name: prefer listLibraries.' };
}

export default listLibraryAlias;
